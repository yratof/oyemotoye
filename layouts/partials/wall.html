{{- /* Image wall: figures are grouped into repeating compositions offset-left (landscape + portrait set lower), solo-right (one landscape), trio (three portraits) */ -}} {{- $figures := partial "figures.html" .page -}} {{- $label := .label -}} {{- if $figures }} {{- /* The first image runs the full width of the browser */ -}} {{- with index $figures 0 }} {{- $small := .img.Resize "1024x webp q80" }} {{- $large := .img.Resize "2048x webp q80" }} {{- end }} {{- $figures = after 1 $figures -}} {{- end }} {{- if $figures }} {{- $kinds := slice (dict "name" "offset-left" "size" 2) (dict "name" "solo-right" "size" 1) (dict "name" "trio" "size" 3) -}} {{- $groups := slice -}} {{- $i := 0 -}} {{- range $n, $_ := $figures -}} {{- if ge $i (len $figures) }}{{ break }}{{ end -}} {{- $kind := index $kinds (mod $n (len $kinds)) -}} {{- $groups = $groups | append (dict "name" $kind.name "items" (first $kind.size (after $i $figures))) -}} {{- $i = add $i $kind.size -}} {{- end }}
{{- range $groups }} {{- $group := . }}
{{- range $k, $f := .items }} {{- $format := "landscape" }} {{- if or (eq $group.name "trio") (and (eq $group.name "offset-left") (eq $k 1)) }}{{ $format = "portrait" }}{{ end }} {{- $small := $f.img.Resize "800x webp q80" }} {{- $large := $f.img.Resize "1600x webp q80" }} {{- end }}
{{- end }}
